#!/bin/bash

# ======================================================
# Скрипт для:
# 1) Очистки локальной истории (оставляет только последний коммит)
# 2) Очистки удалённого репозитория
# 3) Уменьшения размера хранилища
# ======================================================

# --- ЛОКАЛЬНАЯ ОЧИСТКА ---
echo "1. Очищаем локальный репозиторий..."

# Переключаемся на новую ветку без истории
git checkout --orphan temp_clean_branch  # Создаёт ветку без родителя
git add -A                               # Добавляет все файлы в индекс
git commit -m "Initial commit (clean)"   # Создаёт новый корневой коммит

# Удаляем старую ветку main и переименовываем новую
git branch -D main                       # Форсированное удаление старой ветки
git branch -m main                       # Переименовываем temp → main

# --- УДАЛЁННАЯ ОЧИСТКА ---
echo "2. Очищаем удалённый репозиторий..."

# Принудительно отправляем изменения (перезаписываем историю)
git push -f origin main                 # --force заменяет всю историю

# --- ОПТИМИЗАЦИЯ РАЗМЕРА ---
echo "3. Оптимизируем размер хранилища..."

# Локальная оптимизация
git reflog expire --expire=now --all     # Удаляет старые ссылки
git gc --aggressive --prune=now          # Глубокая очистка мусора

# Для удалённого репозитория (если есть SSH-доступ)
REMOTE_REPO="root@192.168.1.50:repository/nixos"
ssh ${REMOTE_REPO%:*} "cd ${REMOTE_REPO#*:} && \
  git reflog expire --expire=now --all && \
  git gc --aggressive --prune=now"       # То же самое на сервере

# --- ПРОВЕРКА ---
echo "4. Проверяем результат:"
echo "Локальный размер: $(du -sh .git | cut -f1)"
echo "Локальные коммиты:"
git log --oneline --all                  # Должен быть 1 коммит

echo "Удалённые коммиты:"
git ls-remote origin                     # Проверяем удалённую историю

echo "Готово! Репозиторий очищен."
